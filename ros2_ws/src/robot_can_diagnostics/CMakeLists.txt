cmake_minimum_required(VERSION 3.8)
project(robot_can_diagnostics)

find_package(ament_cmake REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(mks_servo_can REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(robot_control_interfaces REQUIRED)

add_library(${PROJECT_NAME} SHARED src/diagnostics.cpp src/program_api.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} diagnostic_msgs mks_servo_can rclcpp rclcpp_action std_srvs std_msgs sensor_msgs robot_control_interfaces)
target_link_libraries(${PROJECT_NAME} mks_servo_can::mks_servo_can OpenSSL::Crypto)
add_executable(can_diagnostics src/main.cpp)
target_link_libraries(can_diagnostics ${PROJECT_NAME})
ament_target_dependencies(can_diagnostics rclcpp)
install(DIRECTORY include/ DESTINATION include)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(OpenSSL diagnostic_msgs mks_servo_can rclcpp rclcpp_action std_srvs std_msgs sensor_msgs robot_control_interfaces)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_preview test/test_preview.cpp)
  target_link_libraries(test_preview ${PROJECT_NAME})
  ament_add_gtest(test_reset test/test_reset.cpp)
  target_link_libraries(test_reset ${PROJECT_NAME})
  set_tests_properties(test_reset PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=187;ROS_LOCALHOST_ONLY=1")
endif()

install(TARGETS can_diagnostics DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})

ament_package()
