cmake_minimum_required(VERSION 3.8)
project(robot_motion_planner)
find_package(ament_cmake REQUIRED)
find_package(robot_kinematics REQUIRED)
find_package(mks_servo_can REQUIRED)
find_package(robot_control_interfaces REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(srdfdom REQUIRED)
find_package(urdf REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/planner.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} robot_kinematics mks_servo_can robot_control_interfaces moveit_core moveit_msgs builtin_interfaces srdfdom urdf trajectory_msgs geometry_msgs)
target_link_libraries(${PROJECT_NAME} robot_kinematics::robot_kinematics mks_servo_can::mks_servo_can)
add_executable(planner src/node.cpp)
target_link_libraries(planner ${PROJECT_NAME})
ament_target_dependencies(planner rclcpp rclcpp_action std_msgs robot_control_interfaces)
install(DIRECTORY include/ DESTINATION include)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
add_executable(tcp_jog_replay src/tcp_jog_replay.cpp)
target_link_libraries(tcp_jog_replay robot_kinematics::robot_kinematics)
install(TARGETS tcp_jog_replay DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS tools/tcp_jog_capture.py DESTINATION lib/${PROJECT_NAME} RENAME tcp_jog_capture)
install(TARGETS planner DESTINATION lib/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(robot_kinematics mks_servo_can robot_control_interfaces moveit_core moveit_msgs builtin_interfaces srdfdom urdf geometry_msgs trajectory_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_planner test/test_planner.cpp)
  target_link_libraries(test_planner ${PROJECT_NAME})
  target_compile_definitions(test_planner PRIVATE DEMO_URDF="${CMAKE_CURRENT_SOURCE_DIR}/../robot_description/urdf/preview_6dof.example.urdf")
  ament_add_gtest(test_lin_executor test/test_lin_executor.cpp)
  target_link_libraries(test_lin_executor ${PROJECT_NAME})
  target_compile_definitions(test_lin_executor PRIVATE DEMO_URDF="${CMAKE_CURRENT_SOURCE_DIR}/../robot_description/urdf/preview_6dof.example.urdf")
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_planner_ros test/test_planner_ros.py TIMEOUT 60)
endif()
ament_package()
